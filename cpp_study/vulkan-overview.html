for index, family in enumerate(queue_families):      # C++: for (uint32_t index = 0; ...; ++index)

        return index                                   # optional에 값이 들어감
return None                                            # std::nullopt

4. Logical device: CreateInfo와 포인터 (04장)

Vulkan의 …CreateInfo 구조체는 객체 생성에 필요한 정보를 담습니다. 배열이나 다른 구조체처럼 크기가 정해지지 않은 데이터는 필드에 직접 넣지 못하므로, p로 시작하는 필드(pointer)에 그 데이터의 주소를 넣습니다. 04장에서는 포인터가 두 단계로 이어집니다.

queue priority 1.0f는 같은 device 안의 queue들 사이에서 어느 queue의 작업을 먼저 처리할지에 대한 힌트입니다 (0.0~1.0). queue가 1개뿐이라 비교 대상이 없지만, 규칙상 queue마다 값을 하나씩 꼭 줘야 해서 넣습니다.

7. 정리 순서: RAII

RAII(Resource Acquisition Is Initialization, 자원 획득은 초기화)는 "변수가 사라질 때 자원도 자동으로 정리"하는 C++ 방식입니다. vk::raii:: 타입이 그래서 destroy를 직접 부르지 않아도 됩니다. 멤버 변수는 선언한 순서대로 만들어지고, 반대 순서로 정리됩니다. 그래서 부모(device)가 자식(buffer)보다 먼저 사라지는 일이 없습니다.

예: main() 안쪽 { } 블록이 끝나면 app이 사라지면서 descriptorSets → … → instance → context 순서로 자동 정리되고, 그 뒤에 "… destroyed."가 출력됩니다. BufferResource 안에서도 memory를 먼저 선언했으니 buffer가 먼저, memory가 나중에 정리됩니다.

9. 00~07장을 파이썬 한 덩어리로

# 01: Context와 Instance
context  = Context()
instance = Instance(context, app_name="07-descriptor-sets", api=(1, 3),
                    layers=["VK_LAYER_KHRONOS_validation"],      # 02: validation layer
                    extensions=["VK_EXT_debug_utils"])
debug_messenger = instance.create_debug_messenger(callback=print_to_stderr)

# 03: physical device 선택 + compute queue family 찾기
gpu = next(d for d in instance.physical_devices() if d.api >= (1, 3))   # Tesla T4
family = next(i for i, f in enumerate(gpu.queue_families()) if "compute" in f.flags)  # 0

# 04: logical device 생성 + queue 가져오기
device = Device(gpu, queues=[{"family": family, "count": 1, "priorities": [1.0]}])
compute_queue = device.get_queue(family, 0)

# 05: buffer + device memory 생성, 값 쓰고 검증
inputA, inputB, output = (device.create_buffer(32, memory="host_visible|coherent") for _ in range(3))
inputA.write([0, 1, 2, 3, 4, 5, 6, 7]); assert inputA.read() == [0, 1, 2, 3, 4, 5, 6, 7]

# 06: descriptor set layout
layout = {0: "storage", 1: "storage", 2: "storage"}
# 07: descriptor pool → set 할당 → binding마다 buffer descriptor 갱신
pool = Pool(max_sets=1, storage=3)
descriptor_set = pool.allocate(layout)
descriptor_set.update({0: inputA, 1: inputB, 2: output})

# 08~: 셰이더 → 파이프라인 → 커맨드 버퍼 → 큐에 제출 → GPU가 output = inputA + inputB
